namespace App\Monitoring;

use App\Monitoring\Exceptions\DriverUnavailableException;
use Illuminate\Contracts\Container\Container;

/**
 * Resolves the active driver from BANDWIDTH_MONITORING_DRIVER. There is deliberately
 * no fallback: if the configured driver cannot be used, callers get an exception
 * instead of silently receiving a different (e.g. simulated) driver.
 */
class MonitoringManager
{
    public function __construct(private readonly Container $container) {}

    public function driverName(): string
    {
        return (string) config('campusnet.driver', 'mock');
    }

    public function isDemo(): bool
    {
        return $this->driverName() === 'mock';
    }

    public function isKnownDriver(): bool
    {
        return in_array($this->driverName(), (array) config('campusnet.drivers', []), true);
    }

    /**
     * @throws DriverUnavailableException
     */
    public function driver(): MonitoringDriver
    {
        return match ($this->driverName()) {
            'mock' => $this->container->make(MockMonitoringDriver::class),
            'snmp' => $this->container->make(SnmpMonitoringDriver::class),
            'mikrotik' => throw new DriverUnavailableException('The MikroTik RouterOS driver is not available yet. Use BANDWIDTH_MONITORING_DRIVER=snmp for real devices.'),
            default => throw new DriverUnavailableException('BANDWIDTH_MONITORING_DRIVER has an unknown value. Use mock or snmp.'),
        };
    }

<?php

declare(strict_types=1);

namespace App\Monitoring;

use App\Enums\DeviceStatus;

/**
 * Outcome of testing one configured device.
 *
 * `status` is what the test learned about the device. It is null when the test could
 * not be performed at all (extension missing, no credentials, simulated driver): in
 * that case nothing is known about the device, so its stored status must not change.
 */
final readonly class ConnectionTestResult
{
    /**
     * @param  array<string, scalar|null>  $details
     */
    public function __construct(
        public bool $success,
        public ?DeviceStatus $status,
        public string $message,
        public ?int $latencyMs = null,
        public array $details = [],
        public bool $simulated = false,
    ) {}

    /**
     * @param  array<string, scalar|null>  $details
     */
    public static function reached(string $message, int $latencyMs, array $details = []): self
    {
        return new self(true, DeviceStatus::Online, $message, $latencyMs, $details);
    }

    /** The device was contacted (or could not be contacted) and monitoring is not possible. */
    public static function unreachable(string $message): self
    {
        return new self(false, DeviceStatus::Unreachable, $message);
    }

    /** A setup problem prevented the test. Says nothing about the device itself. */
    public static function notPerformed(string $message): self
    {
        return new self(false, null, $message);
    }

    /** Demo mode: no packet was sent. */
    public static function demo(string $message): self
    {
        return new self(true, null, $message, null, [], true);
    }

    public function shouldPersist(): bool
    {
        return ! $this->simulated && $this->status !== null;
    }

    /**
     * @return array<string, mixed>
     */
    public function toArray(): array
    {
        return [
            'success' => $this->success,
            'status' => $this->status?->value,
            'message' => $this->message,
            'latency_ms' => $this->latencyMs,
            'details' => $this->details,
            'simulated' => $this->simulated,
        ];
    }
}

<?php

declare(strict_types=1);

namespace App\Support;

final class Bandwidth
{
    /**
     * Utilization of the configured internet capacity, in percent.
     *
     * Internet links are full duplex, so the capacity applies to each direction
     * separately; the busiest direction is what saturates the link. The value is
     * not clamped at 100: a result above 100 means the capacity setting is lower
     * than what the link actually carries, which an administrator should notice.
     */
    public static function utilizationPercent(float $inMbps, float $outMbps, float $capacityMbps): ?float
    {
        if ($capacityMbps <= 0) {
            return null;
        }

        return round(max($inMbps, $outMbps, 0.0) / $capacityMbps * 100, 2);
    }

<?php

declare(strict_types=1);

namespace App\Services;

use App\Models\MonitoringSetting;
use App\Models\User;
use Illuminate\Database\QueryException;
use Illuminate\Support\Facades\Cache;
use Illuminate\Support\Facades\Log;
use InvalidArgumentException;

/**
 * Typed access to monitoring_settings with config('campusnet.defaults')
 * as the fallback. The settings screen (a later phase) writes through put().
 */
class SettingsService
{
    private const CACHE_KEY = 'campusnet.settings';

    /**
     * @return array<string, mixed>
     */
    public function all(): array
    {
        $defaults = (array) config('campusnet.defaults');

        try {
            $stored = Cache::remember(
                self::CACHE_KEY,
                now()->addMinutes(10),
                fn (): array => MonitoringSetting::query()->pluck('value', 'key')->all(),
            );
        } catch (QueryException $e) {
            // Table missing (migrations not run yet) or database down: fall back to
            // config defaults so the login page still renders, and say why in the log.
            Log::warning('Monitoring settings unavailable, using defaults.', [
                'exception' => $e::class,
            ]);

            $stored = [];
        }

        $merged = $defaults;
        foreach ($defaults as $key => $default) {
            if (array_key_exists($key, $stored) && $stored[$key] !== null) {
                $merged[$key] = $this->cast($stored[$key], $default);
            }
        }

        return $merged;
    }

    public function get(string $key): mixed
    {
        $all = $this->all();

        if (! array_key_exists($key, $all)) {
            throw new InvalidArgumentException("Unknown setting [{$key}].");
        }

        return $all[$key];
    }

    public function put(string $key, mixed $value, ?User $by = null): void
    {
        if (! array_key_exists($key, (array) config('campusnet.defaults'))) {
            throw new InvalidArgumentException("Unknown setting [{$key}].");
        }

        MonitoringSetting::updateOrCreate(
            ['key' => $key],
            ['value' => is_bool($value) ? (string) (int) $value : (string) $value, 'updated_by' => $by?->getKey()],
        );

        $this->flush();
    }

    public function flush(): void
    {
        Cache::forget(self::CACHE_KEY);
    }

    private function cast(string $stored, mixed $default): mixed
    {
        return match (true) {
            is_int($default) => (int) $stored,
            is_float($default) => (float) $stored,
            is_bool($default) => (bool) (int) $stored,
            default => $stored,
        };
    }
}

<?php

declare(strict_types=1);

namespace App\Services;

use App\Models\MonitoredDevice;
use App\Models\User;
use App\Monitoring\ConnectionTestResult;
use App\Monitoring\Exceptions\DriverUnavailableException;
use App\Monitoring\MonitoringManager;
use Illuminate\Support\Facades\DB;
use Illuminate\Support\Str;
use Throwable;

/**
 * Device lifecycle operations. Every change is written in a transaction together
 * with its audit record; audit metadata never contains credential values.
 */
class MonitoredDeviceService
{
    public function __construct(
        private readonly AuditLogger $audit,
        private readonly MonitoringManager $monitoring,
    ) {}

    /**
     * @param  array<string, mixed>  $data  validated attributes, optionally with `credentials`
     */
    public function create(array $data, User $actor): MonitoredDevice
    {
        return DB::transaction(function () use ($data, $actor): MonitoredDevice {
            $device = new MonitoredDevice($data);
            $device->save();
            $device->refresh(); // load column defaults such as last_status = unknown

            $this->audit->record('device.created', $device, [
                'name' => $device->name,
                'management_ip' => $device->management_ip,
                'device_type' => $device->device_type->value,
                'snmp_version' => $device->snmp_version->value,
                'enabled' => $device->is_enabled,
            ], "Device {$device->name} created.", $actor);

            return $device;
        });
    }

    /**
     * @param  array<string, mixed>  $data  validated attributes; `credentials` only when replaced
     */
    public function update(MonitoredDevice $device, array $data, User $actor): MonitoredDevice
    {
        return DB::transaction(function () use ($device, $data, $actor): MonitoredDevice {
            $device->fill($data);
            $authChanged = array_key_exists('credentials', $data);

            $changes = [];
            foreach ($device->getDirty() as $field => $value) {
                if (in_array($field, ['credentials', 'updated_at'], true)) {
                    continue;
                }

                $changes[$field] = [
                    'from' => $this->forAudit($device->getRawOriginal($field)),
                    'to' => $this->forAudit($value),
                ];
            }

            $device->save();

            if ($changes !== [] || $authChanged) {
                $this->audit->record('device.updated', $device, [
                    'changes' => $changes,
                    'snmp_auth_changed' => $authChanged,
                ], "Device {$device->name} updated.", $actor);
            }

            return $device;
        });
    }

    public function setEnabled(MonitoredDevice $device, bool $enabled, User $actor): MonitoredDevice
    {
        if ($device->is_enabled === $enabled) {
            return $device;
        }

        return DB::transaction(function () use ($device, $enabled, $actor): MonitoredDevice {
            $device->forceFill(['is_enabled' => $enabled])->save();

            $this->audit->record(
                $enabled ? 'device.enabled' : 'device.disabled',
                $device,
                ['name' => $device->name],
                'Device '.$device->name.($enabled ? ' enabled.' : ' disabled.'),
                $actor,
            );

            return $device;
        });
    }

    /**
     * Soft delete. Interfaces and traffic history are kept; the foreign keys restrict
     * any hard delete while history exists.
     */
    public function archive(MonitoredDevice $device, User $actor): void
    {
        DB::transaction(function () use ($device, $actor): void {
            $device->delete();

            $this->audit->record('device.archived', $device, ['name' => $device->name], "Device {$device->name} archived.", $actor);
        });
    }

    /**
     * Test the device using only its stored address, with the active driver.
     * The device's status is updated only when the test actually learned something.
     */
    public function testConnection(MonitoredDevice $device, User $actor): ConnectionTestResult
    {
        try {
            $result = $this->monitoring->driver()->testConnection($device);
        } catch (DriverUnavailableException $e) {
            $result = ConnectionTestResult::notPerformed($e->getMessage());
        } catch (Throwable $e) {
            report($e);

            $result = ConnectionTestResult::notPerformed('The connection test failed unexpectedly. Check the application log.');
        }

        if ($result->shouldPersist()) {
            $device->forceFill([
                'last_polled_at' => now(),
                'last_status' => $result->status,
                'last_successful_poll_at' => $result->success ? now() : $device->last_successful_poll_at,
                'last_error' => $result->success ? null : Str::limit($result->message, 500, ''),
            ])->save();
        }

        $this->audit->record('device.connection_tested', $device, [
            'driver' => $this->monitoring->driverName(),
            'success' => $result->success,
            'status' => $result->status?->value,
            'latency_ms' => $result->latencyMs,
            'simulated' => $result->simulated,
        ], "Connection test run for {$device->name}.", $actor);

        return $result;
    }

    private function forAudit(mixed $value): mixed
    {
        return is_string($value) ? Str::limit($value, 200) : $value;
    }
}

<?php

declare(strict_types=1);

namespace App\Providers;

use App\Enums\UserRole;
use App\Models\User;
use App\Services\AuditLogger;
use App\Services\SettingsService;
use Illuminate\Auth\Events\Failed;
use Illuminate\Auth\Events\Login;
use Illuminate\Auth\Events\Logout;
use Illuminate\Support\Facades\Event;
use Illuminate\Support\Facades\Gate;
use Illuminate\Support\ServiceProvider;

class CampusNetServiceProvider extends ServiceProvider
{
    public function register(): void
    {
        $this->app->singleton(AuditLogger::class);
        $this->app->singleton(SettingsService::class);
    }

    public function boot(): void
    {
        $this->registerPermissionGates();
        $this->registerAuditListeners();
    }

    /**
     * One gate per permission string. Authorize on the backend with
     * `->middleware('can:devices.manage')` or `$this->authorize('devices.manage')`.
     * Hiding a button in React is never the only protection.
     */
    private function registerPermissionGates(): void
    {
        foreach (UserRole::allPermissions() as $permission) {
            Gate::define($permission, fn (User $user): bool => $user->hasPermission($permission));
        }
    }

    private function registerAuditListeners(): void
    {
        Event::listen(Login::class, function (Login $event): void {
            $user = $event->user;

            if (! $user instanceof User) {
                return;
            }

            $user->forceFill(['last_login_at' => now()])->saveQuietly();

            app(AuditLogger::class)->record(
                action: 'auth.login',
                subject: $user,
                description: 'User signed in.',
                actor: $user,
            );
        });

        Event::listen(Logout::class, function (Logout $event): void {
            $user = $event->user;

            if (! $user instanceof User) {
                return;
            }

            app(AuditLogger::class)->record(
                action: 'auth.logout',
                subject: $user,
                description: 'User signed out.',
                actor: $user,
            );
        });

        Event::listen(Failed::class, function (Failed $event): void {
            // Only the attempted e-mail is recorded. The password is never read here.
            app(AuditLogger::class)->record(
                action: 'auth.login_failed',
                subject: $event->user instanceof User ? $event->user : null,
                metadata: ['email' => $event->credentials['email'] ?? null],
                description: 'Failed sign-in attempt.',
            );
        });
    }

    private function purge(AuditLogger $audit): int
    {
        $deviceIds = MonitoredDevice::withTrashed()->where('is_demo', true)->pluck('id');

        if ($deviceIds->isEmpty()) {
            $this->info('There is no demo data to remove.');

            return self::SUCCESS;
        }

        if (! $this->option('force') && ! $this->confirm("Delete {$deviceIds->count()} demo devices with their interfaces, samples and alerts?")) {
            $this->line('Nothing was deleted.');

            return self::SUCCESS;
        }

        $counts = [];

        DB::transaction(function () use ($deviceIds, &$counts): void {
            $counts['samples'] = TrafficSample::whereIn('monitored_device_id', $deviceIds)->delete();
            $counts['alerts'] = NetworkAlert::whereIn('monitored_device_id', $deviceIds)->delete();
            $counts['interfaces'] = NetworkInterface::whereIn('monitored_device_id', $deviceIds)->delete();
            $counts['devices'] = MonitoredDevice::withTrashed()->whereIn('id', $deviceIds)->forceDelete();
        });

        $audit->record('demo.purged', null, $counts, 'Demo data removed.');

        $this->info("Removed {$counts['devices']} demo devices, {$counts['interfaces']} interfaces, {$counts['samples']} samples and {$counts['alerts']} alerts.");

        return self::SUCCESS;
    }

    private function createSamples(
        int $deviceId,
        int $interfaceId,
        InterfaceRole $role,
        float $weight,
        float $capacity,
        string $timezone,
        CarbonImmutable $now,
        int $days,
        ?CarbonImmutable $until,
    ): int {
        $rows = [];
        $total = 0;

        foreach ($this->timestamps($now, $days, $until) as [$at, $step]) {
            [$in, $out] = $this->traffic($role, $weight, $capacity, $at->setTimezone($timezone));
            $stamp = $at->utc()->format('Y-m-d H:i:s');

            $rows[] = [
                'monitored_device_id' => $deviceId,
                'network_interface_id' => $interfaceId,
                'sampled_at' => $stamp,
                'in_octets' => null,
                'out_octets' => null,
                'in_mbps' => $in,
                'out_mbps' => $out,
                // Utilization is measured against the internet capacity, so only WAN rows carry it.
                'utilization_percent' => $role === InterfaceRole::Wan ? Bandwidth::utilizationPercent($in, $out, $capacity) : null,
                'elapsed_seconds' => $step,
                'source' => 'mock',
                'status' => 'ok',
                'is_successful' => true,
                'error_message' => null,
                'created_at' => $stamp,
            ];

            if (count($rows) === 500) {
                DB::table('traffic_samples')->insert($rows);
                $total += count($rows);
                $rows = [];
            }
        }

        if ($rows !== []) {
            DB::table('traffic_samples')->insert($rows);
            $total += count($rows);
        }

        return $total;
    }

    /**
     * Five-minute samples for the history, one-minute samples for the last two hours.
     *
     * @return Generator<int, array{0: CarbonImmutable, 1: int}>
     */
    private function timestamps(CarbonImmutable $now, int $days, ?CarbonImmutable $until): Generator
    {
        $cursor = $now->subDays($days);
        $dense = $now->subHours(2);
        $end = $until ?? $now;

        while ($cursor <= $end) {
            $step = $cursor < $dense ? 300 : 60;

            yield [$cursor, $step];

            $cursor = $cursor->addSeconds($step);
        }
    }

    /**
     * @return array{0: float, 1: float} inbound and outbound Mbps
     */
    private function traffic(InterfaceRole $role, float $weight, float $capacity, CarbonImmutable $local): array
    {
        $load = DemoTraffic::loadFactor((int) $local->format('G'), (int) $local->format('N'));
        $noise = 0.85 + 0.30 * (mt_rand() / mt_getrandmax());
        $ratio = mt_rand() / mt_getrandmax();

        if ($role === InterfaceRole::Wan) {
            $in = $capacity * (0.06 + 0.84 * $load) * $noise;
            $out = $in * (0.22 + 0.15 * $ratio);
        } else {
            $in = $capacity * $weight * (0.02 + 0.40 * $load) * $noise;
            $out = $in * (0.55 + 0.25 * $ratio);
        }

        return [round(min($in, $capacity * 1.2), 4), round(min($out, $capacity * 1.2), 4)];
    }

    private function timezone(string $name): string
    {
        return in_array($name, DateTimeZone::listIdentifiers(), true) ? $name : 'UTC';
    }
