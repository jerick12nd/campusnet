<?php

declare(strict_types=1);

namespace App\Models\Concerns;

use App\Enums\UserRole;

/**
 * Adds CampusNet role and account-status behaviour to the User model.
 *
 * Add `use HasCampusRole;` to App\Models\User. The casts are merged by the
 * initializeHasCampusRole() hook, so the starter kit's own casts() method
 * does not need to be edited.
 *
 * `role` and `is_active` are intentionally NOT mass-assignable. Set them
 * explicitly (e.g. through the campusnet:create-user command or the user
 * management screen) so a crafted request can never promote an account.
 */
trait HasCampusRole
{
    public function initializeHasCampusRole(): void
    {
        $this->mergeCasts([
            'role' => UserRole::class,
            'is_active' => 'boolean',
            'last_login_at' => 'datetime',
        ]);

        // Least-privilege in-memory defaults for brand-new instances. They match the
        // column defaults, so a factory-created user is a viewer and active even
        // before it is reloaded. Models hydrated from the database overwrite these.
        $this->attributes['role'] ??= UserRole::Viewer->value;
        $this->attributes['is_active'] ??= true;
    }

    public function hasRole(UserRole ...$roles): bool
    {
        return in_array($this->role, $roles, true);
    }

    public function isSuperAdmin(): bool
    {
        return $this->role === UserRole::SuperAdmin;
    }

    public function hasPermission(string $permission): bool
    {
        return $this->is_active && $this->role instanceof UserRole && $this->role->can($permission);
    }

    /**
     * @return list<string>
     */
    public function permissions(): array
    {
        if (! $this->is_active || ! $this->role instanceof UserRole) {
            return [];
        }

        return $this->role->permissions();
    }
}

namespace App\Services;

use App\Models\AuditLog;
use App\Models\User;
use Illuminate\Database\Eloquent\Model;
use Illuminate\Support\Facades\Auth;
use Illuminate\Support\Facades\Log;
use Illuminate\Support\Str;
use Throwable;

class AuditLogger
{
    public const REDACTED = '[REDACTED]';

    /**
     * Record an auditable action.
     *
     * @param  array<string, mixed>  $metadata  Non-secret context only. Keys that
     *                                          look like secrets are redacted anyway.
     */
    public function record(
        string $action,
        ?Model $subject = null,
        array $metadata = [],
        ?string $description = null,
        ?User $actor = null,
    ): ?AuditLog {
        try {
            $actor ??= Auth::user();
            $request = app()->bound('request') ? request() : null;

            return AuditLog::create([
                'user_id' => $actor?->getKey(),
                'actor_name' => $actor?->name,
                'action' => $action,
                'subject_type' => $subject ? $subject::class : null,
                'subject_id' => $subject?->getKey(),
                'description' => $description ? Str::limit($description, 250, '') : null,
                'metadata' => $metadata === [] ? null : $this->redact($metadata),
                'ip_address' => $request?->ip(),
                'user_agent' => $request?->userAgent() ? Str::limit($request->userAgent(), 250, '') : null,
            ]);
        } catch (Throwable $e) {
            // Auditing must never break the user's action, but a failure must
            // not be silent either. Log the reason (no metadata, no secrets).
            Log::error('Audit log write failed', [
                'action' => $action,
                'exception' => $e::class,
            ]);

            return null;
        }
    }

    /**
     * Replace the value of any key that looks like a secret, at any depth.
     *
     * @param  array<array-key, mixed>  $data
     * @return array<array-key, mixed>
     */
    public function redact(array $data): array
    {
        $fragments = array_map('strtolower', (array) config('campusnet.redact_keys', []));

        $clean = [];
        foreach ($data as $key => $value) {
            if (is_string($key) && $this->looksSecret($key, $fragments)) {
                $clean[$key] = self::REDACTED;

                continue;
            }

            $clean[$key] = is_array($value) ? $this->redact($value) : $value;
        }

        return $clean;
    }

    /**
     * @param  list<string>  $fragments
     */
    private function looksSecret(string $key, array $fragments): bool
    {
        $key = strtolower($key);

        foreach ($fragments as $fragment) {
            if ($fragment !== '' && str_contains($key, $fragment)) {
                return true;
            }
        }

        return false;
    }
}
<?php

namespace App\Models;

use App\Concerns\GeneratesUniqueTeamSlugs;
use App\Enums\TeamRole;
use Database\Factories\TeamFactory;
use Illuminate\Database\Eloquent\Attributes\Fillable;
use Illuminate\Database\Eloquent\Collection;
use Illuminate\Database\Eloquent\Factories\HasFactory;
use Illuminate\Database\Eloquent\Model;
use Illuminate\Database\Eloquent\Relations\BelongsToMany;
use Illuminate\Database\Eloquent\Relations\HasMany;
use Illuminate\Database\Eloquent\SoftDeletes;
use Illuminate\Support\Carbon;

/**
 * @property int $id
 * @property string $name
 * @property string $slug
 * @property bool $is_personal
 * @property Carbon|null $created_at
 * @property Carbon|null $updated_at
 * @property Carbon|null $deleted_at
 * @property-read Collection<int, TeamInvitation> $invitations
 * @property-read Collection<int, Membership> $memberships
 * @property-read Collection<int, User> $members
 */
#[Fillable(['name', 'slug', 'is_personal'])]
class Team extends Model
{
    /** @use HasFactory<TeamFactory> */
    use GeneratesUniqueTeamSlugs, HasFactory, SoftDeletes;

    /**
     * Bootstrap the model and its traits.
     */
    protected static function boot(): void
    {
        parent::boot();

        static::creating(function (Team $team) {
            if (empty($team->slug)) {
                $team->slug = static::generateUniqueTeamSlug($team->name);
            }
        });

        static::updating(function (Team $team) {
            if ($team->isDirty('name')) {
                $team->slug = static::generateUniqueTeamSlug($team->name, $team->id);
            }
        });
    }

    /**
     * Get the team owner.
     */
    public function owner(): ?Model
    {
        return $this->members()
            ->wherePivot('role', TeamRole::Owner->value)
            ->first();
    }

    /**
     * Get all members of this team.
     *
     * @return BelongsToMany<User, $this, Membership, 'pivot'>
     */
    public function members(): BelongsToMany
    {
        return $this->belongsToMany(User::class, 'team_members', 'team_id', 'user_id')
            ->using(Membership::class)
            ->withPivot(['role'])
            ->withTimestamps();
    }

    /**
     * Get all memberships for this team.
     *
     * @return HasMany<Membership, $this>
     */
    public function memberships(): HasMany
    {
        return $this->hasMany(Membership::class);
    }

    /**
     * Get all invitations for this team.
     *
     * @return HasMany<TeamInvitation, $this>
     */
    public function invitations(): HasMany
    {
        return $this->hasMany(TeamInvitation::class);
    }

    /**
     * Get the attributes that should be cast.
     *
     * @return array<string, string>
     */
    protected function casts(): array
    {
        return [
            'is_personal' => 'boolean',
        ];
    }

    /**
     * Get the route key for the model.
     */
    public function getRouteKeyName(): string
    {
        return 'slug';
    }
}

namespace App\Actions\Fortify;

use App\Concerns\PasswordValidationRules;
use App\Models\User;
use Illuminate\Support\Facades\Validator;
use Laravel\Fortify\Contracts\ResetsUserPasswords;

class ResetUserPassword implements ResetsUserPasswords
{
    use PasswordValidationRules;

    /**
     * Validate and reset the user's forgotten password.
     *
     * @param  array<string, string>  $input
     */
    public function reset(User $user, array $input): void
    {
        Validator::make($input, [
            'password' => $this->passwordRules(),
        ])->validate();

        $user->forceFill([
            'password' => $input['password'],
        ])->save();
    }
}

<?php

namespace App\Actions\Teams;

use App\Enums\TeamRole;
use App\Models\Team;
use App\Models\User;
use Illuminate\Support\Facades\DB;

class CreateTeam
{
    /**
     * Create a new team and add the user as owner.
     */
    public function handle(User $user, string $name, bool $isPersonal = false): Team
    {
        return DB::transaction(function () use ($user, $name, $isPersonal) {
            $team = Team::create([
                'name' => $name,
                'is_personal' => $isPersonal,
            ]);

            $membership = $team->memberships()->create([
                'user_id' => $user->id,
                'role' => TeamRole::Owner,
            ]);

            $user->switchTeam($team);

            return $team;
        });
    }
}
