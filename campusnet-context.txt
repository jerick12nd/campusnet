namespace App\Monitoring\Drivers;

use App\Enums\DataSource;
use App\Enums\SnmpVersion;
use App\Models\MonitoredDevice;
use App\Monitoring\ConnectionTestResult;
use App\Monitoring\Contracts\MonitoringDriver;
use App\Monitoring\Exceptions\SnmpException;
use App\Monitoring\Snmp\SnmpClient;
use App\Services\SettingsService;
use Illuminate\Support\Str;

/**
 * Real monitoring over SNMP (read-only GET requests).
 */
class SnmpMonitoringDriver implements MonitoringDriver
{
    public function __construct(private readonly SettingsService $settings) {}

    public function source(): DataSource
    {
        return DataSource::Snmp;
    }

    public function testConnection(MonitoredDevice $device): ConnectionTestResult
    {
        if (! SnmpClient::isAvailable()) {
            return ConnectionTestResult::notPerformed(
                'The PHP SNMP extension is not enabled on this server, so real devices cannot be tested. '
                .'Enable extension=snmp in php.ini and restart the web server. Demo mode works without it.'
            );
        }

        if ($device->snmp_version === SnmpVersion::V2c && ! config('campusnet.allow_snmp_v2c', false)) {
            return ConnectionTestResult::notPerformed(
                'SNMPv2c is disabled on this server. Edit the device to use SNMPv3, or set CAMPUSNET_ALLOW_SNMP_V2C=true for an approved, trusted management network.'
            );
        }

        $credentials = $device->credentials;

        if (empty($credentials)) {
            return ConnectionTestResult::notPerformed(
                'No SNMP credentials are stored for this device. Edit the device and add them.'
            );
        }

        $client = new SnmpClient(
            $device->management_ip,
            $device->snmp_port,
            $device->snmp_version,
            $credentials,
            (int) $this->settings->get('polling_timeout_seconds'),
            1,
        );

        $startedAt = hrtime(true);

        try {
            $description = $client->get(SnmpClient::OID_SYS_DESCR);
        } catch (SnmpException $e) {
            return ConnectionTestResult::unreachable($e->getMessage());
        }

        $latencyMs = (int) round((hrtime(true) - $startedAt) / 1_000_000);

        $details = ['sys_descr' => Str::limit($description, 200)];

        // The device already answered, so these extras are best-effort.
        try {
            $details['sys_name'] = Str::limit($client->get(SnmpClient::OID_SYS_NAME), 120);
        } catch (SnmpException) {
        }

        try {
            $ticks = $client->get(SnmpClient::OID_SYS_UPTIME);

            if (ctype_digit($ticks)) {
                $seconds = intdiv((int) $ticks, 100);
                $details['uptime'] = intdiv($seconds, 86400).'d '.intdiv($seconds % 86400, 3600).'h';
            }
        } catch (SnmpException) {
        }

        if ($device->snmp_version === SnmpVersion::V2c) {
            $details['security_note'] = 'SNMPv2c sends the community string unencrypted. Use it only on a trusted management network, or move the device to SNMPv3.';
        }

        return ConnectionTestResult::reached("The device answered the SNMP request in {$latencyMs} ms.", $latencyMs, $details);
    }
}

<?php

declare(strict_types=1);

namespace App\Monitoring\Snmp;

use App\Enums\SnmpVersion;
use App\Monitoring\Exceptions\SnmpException;

/**
 * Thin, read-only wrapper around the PHP SNMP extension (GET requests only).
 *
 * Credentials are used to build the request and are never logged or put in
 * exception messages: failures are translated into fixed, safe sentences.
 */
final class SnmpClient
{
    public const OID_SYS_DESCR = '1.3.6.1.2.1.1.1.0';

    public const OID_SYS_UPTIME = '1.3.6.1.2.1.1.3.0';

    public const OID_SYS_NAME = '1.3.6.1.2.1.1.5.0';

    /**
     * @param  array<string, mixed>  $credentials
     */
    public function __construct(
        private readonly string $host,
        private readonly int $port,
        private readonly SnmpVersion $version,
        private readonly array $credentials,
        private readonly int $timeoutSeconds = 5,
        private readonly int $retries = 1,
    ) {}

    public static function isAvailable(): bool
    {
        return extension_loaded('snmp') && function_exists('snmp2_get') && function_exists('snmp3_get');
    }

    /**
     * Fetch one scalar OID.
     *
     * @throws SnmpException
     */
    public function get(string $oid): string
    {
        $timeout = max(1, $this->timeoutSeconds) * 1_000_000; // the extension expects microseconds
        $retries = max(0, $this->retries);
        $target = $this->target();

        snmp_set_valueretrieval(SNMP_VALUE_PLAIN);

        $warning = null;
        set_error_handler(static function (int $severity, string $message) use (&$warning): bool {
            $warning = $message;

            return true;
        });

        try {
            $value = match ($this->version) {
                SnmpVersion::V2c => snmp2_get(
                    $target,
                    (string) ($this->credentials['community'] ?? ''),
                    $oid,
                    $timeout,
                    $retries,
                ),
                SnmpVersion::V3 => snmp3_get(
                    $target,
                    (string) ($this->credentials['username'] ?? ''),
                    (string) ($this->credentials['security_level'] ?? 'authPriv'),
                    (string) ($this->credentials['auth_protocol'] ?? 'SHA'),
                    (string) ($this->credentials['auth_password'] ?? ''),
                    (string) ($this->credentials['priv_protocol'] ?? ''),
                    (string) ($this->credentials['priv_password'] ?? ''),
                    $oid,
                    $timeout,
                    $retries,
                ),
            };
        } finally {
            restore_error_handler();
        }

        if ($value === false) {
            throw new SnmpException(self::describeFailure($warning));
        }

        return trim((string) $value);
    }

    /**
     * Turn a raw PHP/net-snmp warning into a fixed sentence. The raw text is discarded.
     */
    public static function describeFailure(?string $rawWarning): string
    {
        $raw = strtolower((string) $rawWarning);

        return match (true) {
            str_contains($raw, 'no response') || str_contains($raw, 'timeout') => 'No response from the device (timeout). Check the IP address, SNMP port, firewall rules and that SNMP is enabled on the device. With SNMPv2c a wrong community string also looks like a timeout.',
            str_contains($raw, 'authentication')
                || str_contains($raw, 'unknown user')
                || str_contains($raw, 'decrypt')
                || str_contains($raw, 'security level')
                || str_contains($raw, 'engine') => 'SNMP authentication failed. Check the SNMPv3 user name, security level, protocols and passphrases.',
            str_contains($raw, 'unknown host') || str_contains($raw, 'name or service not known') => 'The device address could not be resolved.',
            default => 'The SNMP request failed. Check the device address, port and credentials.',
        };
    }

    private function target(): string
    {
        return str_contains($this->host, ':')
            ? "[{$this->host}]:{$this->port}"
            : "{$this->host}:{$this->port}";
    }
}

<?php

declare(strict_types=1);

namespace App\Services;

use App\Enums\DeviceStatus;
use App\Enums\SampleStatus;
use App\Models\MonitoredDevice;
use App\Models\NetworkAlert;
use App\Models\NetworkInterface;
use App\Models\TrafficSample;
use App\Monitoring\MonitoringManager;
use App\Support\Bandwidth;
use App\Support\SqlDialect;
use Carbon\CarbonImmutable;
use Illuminate\Database\Eloquent\Builder;
use Illuminate\Support\Collection;
use Illuminate\Support\Facades\DB;

/**
 * Builds the dashboard payload from stored data only. Nothing here is hardcoded.
 *
 * Honesty rules:
 *  - Only samples whose `source` equals the active driver are used, so simulated
 *    (mock) samples can never appear once a real driver is selected, and vice versa.
 *  - Demo devices (is_demo) are ignored unless the mock driver is active.
 *  - "Current" values come from the latest valid sample and are flagged stale when
 *    that sample is older than the configured stale-data window.
 */
final class DashboardService
{
    public const DEFAULT_RANGE = '24h';

    /** Range length and chart bucket size, in seconds. */
    public const RANGES = [
        '1h' => ['seconds' => 3600, 'bucket' => 60],
        '24h' => ['seconds' => 86400, 'bucket' => 900],
        '7d' => ['seconds' => 604800, 'bucket' => 3600],
        '30d' => ['seconds' => 2592000, 'bucket' => 21600],
    ];

    public function __construct(
        private readonly SettingsService $settings,
        private readonly MonitoringManager $monitoring,
    ) {}

    /**
     * @return array<string, mixed>
     */
    public function build(string $range = self::DEFAULT_RANGE, ?CarbonImmutable $now = null): array
    {
        $range = isset(self::RANGES[$range]) ? $range : self::DEFAULT_RANGE;
        $config = self::RANGES[$range];
        $now ??= CarbonImmutable::now();
        $from = $now->subSeconds($config['seconds']);

        $settings = $this->settings->all();
        $source = $this->monitoring->driverName();
        $capacity = (float) $settings['bandwidth_capacity_mbps'];
        $staleAfter = $now->subMinutes((int) $settings['stale_data_minutes']);

        $wanInterfaces = $this->wanInterfaces($source);
        $devices = $this->deviceCounts($source);
        $latestAt = $this->latestMeasurementAt($source);
        $history = $this->history($wanInterfaces->pluck('id')->all(), $source, $from, $config['bucket'], $capacity);

        return [
            'range' => $range,
            'ranges' => array_keys(self::RANGES),
            'mode' => ['demo' => $this->monitoring->isDemo(), 'driver' => $source],
            'capacity_mbps' => $capacity,
            'thresholds' => [
                'warning' => (float) $settings['warning_threshold_percent'],
                'critical' => (float) $settings['critical_threshold_percent'],
            ],
            'refresh_seconds' => (int) $settings['dashboard_refresh_seconds'],
            'generated_at' => $now->toIso8601String(),
            'collector' => $this->collector($devices['total'], $latestAt, $staleAfter),
            'latest_measurement_at' => $latestAt?->toIso8601String(),
            'wan' => $this->wan($wanInterfaces, $source, $capacity, $staleAfter),
            'devices' => $devices,
            'alerts' => $this->alertCounts($source),
            'history' => $history['points'],
            'summary' => $history['summary'],
            'top_interfaces' => $this->topInterfaces($source, $from),
            'incidents' => $this->incidents($source),
        ];
    }

    /**
     * @return Builder<MonitoredDevice>
     */
    private function deviceQuery(string $source): Builder
    {
        return MonitoredDevice::query()
            ->when($source !== 'mock', fn (Builder $query) => $query->where('is_demo', false));
    }

    /**
     * @return Builder<NetworkAlert>
     */
    private function alertQuery(string $source): Builder
    {
        return NetworkAlert::query()->where(function (Builder $query) use ($source): void {
            $query->whereNull('monitored_device_id')
                ->orWhereHas('device', fn (Builder $device) => $device
                    ->when($source !== 'mock', fn (Builder $q) => $q->where('is_demo', false)));
        });
    }

    /**
     * @return Collection<int, NetworkInterface>
     */
    private function wanInterfaces(string $source): Collection
    {
        return NetworkInterface::query()
            ->with('device:id,name')
            ->wan()
            ->monitored()
            ->whereHas('device', fn (Builder $device) => $device
                ->where('is_enabled', true)
                ->when($source !== 'mock', fn (Builder $q) => $q->where('is_demo', false)))
            ->orderBy('id')
            ->get();
    }

    /**
     * @return array{total: int, online: int, unreachable: int, unknown: int, disabled: int}
     */
    private function deviceCounts(string $source): array
    {
        /** @var Collection<string, int|string> $byStatus */
        $byStatus = $this->deviceQuery($source)
            ->where('is_enabled', true)
            ->selectRaw('last_status, COUNT(*) as aggregate')
            ->groupBy('last_status')
            ->toBase()
            ->pluck('aggregate', 'last_status');

        $online = (int) ($byStatus[DeviceStatus::Online->value] ?? 0);
        $unreachable = (int) ($byStatus[DeviceStatus::Unreachable->value] ?? 0);
        $unknown = (int) ($byStatus[DeviceStatus::Unknown->value] ?? 0);

        return [
            'total' => $online + $unreachable + $unknown,
            'online' => $online,
            'unreachable' => $unreachable,
            'unknown' => $unknown,
            'disabled' => $this->deviceQuery($source)->where('is_enabled', false)->count(),
        ];
    }

    private function latestMeasurementAt(string $source): ?CarbonImmutable
    {
        $sample = TrafficSample::query()
            ->where('source', $source)
            ->where('is_successful', true)
            ->latest('sampled_at')
            ->first(['sampled_at']);

        return $sample ? CarbonImmutable::instance($sample->sampled_at) : null;
    }

    /**
     * @return array{status: string, message: string|null}
     */
    private function collector(int $enabledDevices, ?CarbonImmutable $latestAt, CarbonImmutable $staleAfter): array
    {
        if (! $this->monitoring->isKnownDriver()) {
            return ['status' => 'misconfigured', 'message' => 'BANDWIDTH_MONITORING_DRIVER has an unknown value. Use mock or snmp.'];
        }

        if ($enabledDevices === 0) {
            return ['status' => 'no_devices', 'message' => 'No devices are being monitored yet. Add a device on the Network Devices page.'];
        }

        if ($latestAt === null) {
            return ['status' => 'no_data', 'message' => 'No measurements have been collected yet.'];
        }

        if ($latestAt < $staleAfter) {
            return ['status' => 'stale', 'message' => 'The last measurement is '.$latestAt->diffForHumans().'. The collector may be offline, so the values below are not current.'];
        }

        return ['status' => 'ok', 'message' => null];
    }

    /**
     * @param  Collection<int, NetworkInterface>  $wanInterfaces
     * @return array<string, mixed>
     */
    private function wan(Collection $wanInterfaces, string $source, float $capacity, CarbonImmutable $staleAfter): array
    {
        $interfaces = $wanInterfaces->map(fn (NetworkInterface $interface): array => [
            'id' => $interface->id,
            'name' => $interface->name,
            'device' => $interface->device?->name,
        ])->values()->all();

        $latest = $wanInterfaces
            ->map(fn (NetworkInterface $interface) => $interface->trafficSamples()
                ->where('source', $source)
                ->withValidRate()
                ->latest('sampled_at')
                ->first())
            ->filter()
            ->values();

        if ($latest->isEmpty()) {
            return ['configured' => $wanInterfaces->isNotEmpty(), 'interfaces' => $interfaces, 'current' => null];
        }

        $in = (float) $latest->sum(fn (TrafficSample $sample): float => (float) $sample->in_mbps);
        $out = (float) $latest->sum(fn (TrafficSample $sample): float => (float) $sample->out_mbps);
        $sampledAt = CarbonImmutable::instance($latest->sortByDesc(fn (TrafficSample $s): int => $s->sampled_at->getTimestamp())->first()->sampled_at);

        return [
            'configured' => true,
            'interfaces' => $interfaces,
            'current' => [
                'in_mbps' => round($in, 2),
                'out_mbps' => round($out, 2),
                'utilization_percent' => Bandwidth::utilizationPercent($in, $out, $capacity),
                'sampled_at' => $sampledAt->toIso8601String(),
                'stale' => $sampledAt < $staleAfter,
                'interfaces_reporting' => $latest->count(),
                'interfaces_total' => $wanInterfaces->count(),
            ],
        ];
    }

    /**
     * Bucketed history of the WAN interfaces. Values of several WAN interfaces are summed
     * per bucket. The per-bucket peak is the sum of each interface's bucket maximum, which
     * is exact for a single WAN link and an upper bound when several links peak at
     * different moments.
     *
     * @param  list<int>  $interfaceIds
     * @return array{points: list<array<string, mixed>>, summary: array<string, float|null>|null}
     */
    private function history(array $interfaceIds, string $source, CarbonImmutable $from, int $bucket, float $capacity): array
    {
        if ($interfaceIds === []) {
            return ['points' => [], 'summary' => null];
        }

        $bucketSql = SqlDialect::bucket('sampled_at', $bucket);

        $rows = DB::table('traffic_samples')
            ->selectRaw("{$bucketSql} as bucket, network_interface_id, AVG(in_mbps) as avg_in, AVG(out_mbps) as avg_out, MAX(in_mbps) as max_in, MAX(out_mbps) as max_out")
            ->whereIn('network_interface_id', $interfaceIds)
            ->where('source', $source)
            ->where('status', SampleStatus::Ok->value)
            ->where('sampled_at', '>=', $from)
            ->groupByRaw('bucket, network_interface_id')
            ->orderByRaw('bucket')
            ->get();

        $points = $rows->groupBy('bucket')->map(function (Collection $group, $bucketStart) use ($capacity): array {
            $in = (float) $group->sum('avg_in');
            $out = (float) $group->sum('avg_out');
            $peakIn = (float) $group->sum('max_in');
            $peakOut = (float) $group->sum('max_out');

            return [
                't' => CarbonImmutable::createFromTimestampUTC((int) $bucketStart)->toIso8601String(),
                'in_mbps' => round($in, 2),
                'out_mbps' => round($out, 2),
                'peak_in_mbps' => round($peakIn, 2),
                'peak_out_mbps' => round($peakOut, 2),
                'utilization_percent' => Bandwidth::utilizationPercent($in, $out, $capacity),
                'peak_utilization_percent' => Bandwidth::utilizationPercent($peakIn, $peakOut, $capacity),
            ];
        })->values();

        if ($points->isEmpty()) {
            return ['points' => [], 'summary' => null];
        }

        $averageUtilization = $points->pluck('utilization_percent')->filter(fn ($value) => $value !== null);
        $peakUtilization = $points->pluck('peak_utilization_percent')->filter(fn ($value) => $value !== null);

        return [
            'points' => $points->all(),
            'summary' => [
                'avg_utilization_percent' => $averageUtilization->isEmpty() ? null : round((float) $averageUtilization->avg(), 2),
                'peak_utilization_percent' => $peakUtilization->isEmpty() ? null : (float) $peakUtilization->max(),
                'avg_in_mbps' => round((float) $points->avg('in_mbps'), 2),
                'avg_out_mbps' => round((float) $points->avg('out_mbps'), 2),
                'peak_in_mbps' => (float) $points->max('peak_in_mbps'),
                'peak_out_mbps' => (float) $points->max('peak_out_mbps'),
            ],
        ];
    }

    /**
     * @return list<array<string, mixed>>
     */
    private function topInterfaces(string $source, CarbonImmutable $from): array
    {
        /** @var Collection<int, object{network_interface_id: int, avg_in: string, avg_out: string}> $rows */
        $rows = DB::table('traffic_samples')
            ->selectRaw('network_interface_id, AVG(in_mbps) as avg_in, AVG(out_mbps) as avg_out')
            ->where('source', $source)
            ->where('status', SampleStatus::Ok->value)
            ->where('sampled_at', '>=', $from)
            ->groupBy('network_interface_id')
            ->orderByRaw('(AVG(in_mbps) + AVG(out_mbps)) DESC')
            ->limit(5)
            ->get();

        if ($rows->isEmpty()) {
            return [];
        }

        $interfaces = NetworkInterface::query()
            ->with(['device' => fn ($query) => $query->withTrashed()->select('id', 'name')])
            ->whereIn('id', $rows->pluck('network_interface_id'))
            ->get()
            ->keyBy('id');

        return $rows->map(function (object $row) use ($interfaces): array {
            $interface = $interfaces->get($row->network_interface_id);
            $in = (float) $row->avg_in;
            $out = (float) $row->avg_out;

            return [
                'id' => (int) $row->network_interface_id,
                'name' => $interface?->name ?? 'Unknown interface',
                'device' => $interface?->device?->name,
                'role' => $interface?->role->label(),
                'avg_in_mbps' => round($in, 2),
                'avg_out_mbps' => round($out, 2),
                'avg_total_mbps' => round($in + $out, 2),
            ];
        })->values()->all();
    }

    /**
     * @return array{active: int, critical: int}
     */
    private function alertCounts(string $source): array
    {
        return [
            'active' => $this->alertQuery($source)->open()->count(),
            'critical' => $this->alertQuery($source)->open()->critical()->count(),
        ];
    }

    /**
     * @return list<array<string, mixed>>
     */
    private function incidents(string $source): array
    {
        return $this->alertQuery($source)
            ->with(['device' => fn ($query) => $query->select('id', 'name')])
            ->latest('triggered_at')
            ->limit(8)
            ->get()
            ->map(fn (NetworkAlert $alert): array => [
                'id' => $alert->id,
                'title' => $alert->title,
                'severity' => $alert->severity->value,
                'status' => $alert->status->value,
                'device' => $alert->device?->name,
                'triggered_at' => $alert->triggered_at->toIso8601String(),
                'resolved_at' => $alert->resolved_at?->toIso8601String(),
            ])->all();
    }
}

<?php

declare(strict_types=1);

namespace App\Services;

use App\Models\AuditLog;
use App\Models\User;
use Illuminate\Database\Eloquent\Model;
use Illuminate\Support\Facades\Auth;
use Illuminate\Support\Facades\Log;
use Illuminate\Support\Str;
use Throwable;

class AuditLogger
{
    public const REDACTED = '[REDACTED]';

    /**
     * Record an auditable action.
     *
     * @param  array<string, mixed>  $metadata  Non-secret context only. Keys that
     *                                          look like secrets are redacted anyway.
     */
    public function record(
        string $action,
        ?Model $subject = null,
        array $metadata = [],
        ?string $description = null,
        ?User $actor = null,
    ): ?AuditLog {
        try {
            $actor ??= Auth::user();
            $request = app()->bound('request') ? request() : null;

            return AuditLog::create([
                'user_id' => $actor?->getKey(),
                'actor_name' => $actor?->name,
                'action' => $action,
                'subject_type' => $subject ? $subject::class : null,
                'subject_id' => $subject?->getKey(),
                'description' => $description ? Str::limit($description, 250, '') : null,
                'metadata' => $metadata === [] ? null : $this->redact($metadata),
                'ip_address' => $request?->ip(),
                'user_agent' => $request?->userAgent() ? Str::limit($request->userAgent(), 250, '') : null,
            ]);
        } catch (Throwable $e) {
            // Auditing must never break the user's action, but a failure must
            // not be silent either. Log the reason (no metadata, no secrets).
            Log::error('Audit log write failed', [
                'action' => $action,
                'exception' => $e::class,
            ]);

            return null;
        }
    }

    /**
     * Replace the value of any key that looks like a secret, at any depth.
     *
     * @param  array<array-key, mixed>  $data
     * @return array<array-key, mixed>
     */
    public function redact(array $data): array
    {
        $fragments = array_map('strtolower', (array) config('campusnet.redact_keys', []));

        $clean = [];
        foreach ($data as $key => $value) {
            if (is_string($key) && $this->looksSecret($key, $fragments)) {
                $clean[$key] = self::REDACTED;

                continue;
            }

            $clean[$key] = is_array($value) ? $this->redact($value) : $value;
        }

        return $clean;
    }

    /**
     * @param  list<string>  $fragments
     */
    private function looksSecret(string $key, array $fragments): bool
    {
        $key = strtolower($key);

        foreach ($fragments as $fragment) {
            if ($fragment !== '' && str_contains($key, $fragment)) {
                return true;
            }
        }

        return false;
    }
}

    /**
     * Typical campus load (0..1) for a local hour (0-23) and ISO weekday (1 = Monday).
     * Quiet at night, a broad peak around 13:00, much lighter on weekends.
     */
    public static function loadFactor(int $hour, int $isoWeekday): float
    {
        $bump = exp(-(($hour + 0.5 - 13.0) ** 2) / (2 * 3.5 ** 2));
        $load = 0.06 + 0.94 * $bump;

        return round($isoWeekday >= 6 ? $load * 0.35 : $load, 4);
    }

            ->withCount([
                'interfaces',
                'alerts as open_alerts_count' => fn ($query) => $query->open(),
            ])
            ->when($filters['search'] ?? null, function ($query, string $term): void {
                $like = '%'.$term.'%';

                $query->where(function ($query) use ($like): void {
                    $query->where('name', 'like', $like)
                        ->orWhere('management_ip', 'like', $like)
                        ->orWhere('vendor', 'like', $like)
                        ->orWhere('model', 'like', $like)
                        ->orWhere('location', 'like', $like);
                });

/** Display helpers. Timestamps arrive as ISO 8601 strings and are shown in the campus time zone. */

export function safeTimeZone(timeZone?: string | null): string {
    if (!timeZone) {
        return 'UTC';
    }

    try {
        new Intl.DateTimeFormat('en', { timeZone });

        return timeZone;
    } catch {
        return 'UTC';
    }
}

export function formatMbps(value: number | null | undefined): string {
    if (value === null || value === undefined || Number.isNaN(value)) {
        return 'â€”';
    }

    return Math.abs(value) >= 1000 ? `${(value / 1000).toFixed(2)} Gbps` : `${value.toFixed(2)} Mbps`;
}

export function formatPercent(value: number | null | undefined, digits = 1): string {
    if (value === null || value === undefined || Number.isNaN(value)) {
        return 'â€”';
    }

    return `${value.toFixed(digits)}%`;
}

export function formatDateTime(iso: string | null | undefined, timeZone?: string | null): string {
    if (!iso) {
        return 'â€”';
    }

    const date = new Date(iso);

    if (Number.isNaN(date.getTime())) {
        return 'â€”';
    }

    return new Intl.DateTimeFormat('en-PH', {
        dateStyle: 'medium',
        timeStyle: 'short',
        timeZone: safeTimeZone(timeZone),
    }).format(date);
}

export function formatClock(iso: string | null | undefined, timeZone?: string | null): string {
    if (!iso) {
        return 'â€”';
    }

    const date = new Date(iso);

    if (Number.isNaN(date.getTime())) {
        return 'â€”';
    }

    return new Intl.DateTimeFormat('en-PH', {
        hour: '2-digit',
        minute: '2-digit',
        second: '2-digit',
        hour12: false,
        timeZone: safeTimeZone(timeZone),
    }).format(date);
}

export function timeAgo(iso: string | null | undefined, now: number = Date.now()): string {
    if (!iso) {
        return 'never';
    }

    const then = new Date(iso).getTime();

    if (Number.isNaN(then)) {
        return 'never';
    }

    const seconds = Math.max(0, Math.round((now - then) / 1000));

    if (seconds < 45) {
        return 'just now';
    }

    if (seconds < 3600) {
        return `${Math.round(seconds / 60)} min ago`;
    }

    if (seconds < 86400) {
        return `${Math.round(seconds / 3600)} h ago`;
    }

    return `${Math.round(seconds / 86400)} d ago`;
}

/** Tick label for the chart time axis: clock time for short ranges, date for long ones. */
export function formatAxisTime(timestamp: number, range: string, timeZone?: string | null): string {
    const options: Intl.DateTimeFormatOptions =
        range === '7d' || range === '30d'
            ? { month: 'short', day: 'numeric' }
            : { hour: '2-digit', minute: '2-digit', hour12: false };

    return new Intl.DateTimeFormat('en-PH', { ...options, timeZone: safeTimeZone(timeZone) }).format(new Date(timestamp));
}

export function rangeLabel(range: string): string {
    return (
        { '1h': 'Last hour', '24h': 'Last 24 hours', '7d': 'Last 7 days', '30d': 'Last 30 days' }[range] ?? range
    );
}
