type Level = 'none' | 'normal' | 'warning' | 'critical';

function levelFor(value: number | null | undefined, thresholds: { warning: number; critical: number }): Level {
    if (value === null || value === undefined) {
        return 'none';
    }

    if (value >= thresholds.critical) {
        return 'critical';
    }

    return value >= thresholds.warning ? 'warning' : 'normal';
}

const LEVELS: Record<Level, { label: string; bar: string }> = {
    none: { label: 'No data', bar: 'bg-zinc-400' },
    normal: { label: 'Normal', bar: 'bg-emerald-500' },
    warning: { label: 'Warning level', bar: 'bg-amber-500' },
    critical: { label: 'Critical level', bar: 'bg-red-500' },
};

function StatCard({ title, value, hint, icon: Icon, children }: { title: string; value: string; hint?: ReactNode; icon: typeof Gauge; children?: ReactNode }) {
    return (
        <section aria-label={title} className="rounded-xl border bg-card p-4 text-card-foreground shadow-xs">
            <div className="flex items-center justify-between gap-2 text-sm text-muted-foreground">
                <h3 className="font-medium">{title}</h3>
                <Icon className="size-4" aria-hidden="true" />
            </div>
            <p className="mt-2 text-2xl font-semibold tabular-nums">{value}</p>
            {hint && <p className="mt-1 text-xs text-muted-foreground">{hint}</p>}
            {children}
        </section>
    );
}
